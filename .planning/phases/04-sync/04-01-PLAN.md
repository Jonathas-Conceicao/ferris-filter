In `src/sync/client.rs`: define the seam (use `async_trait` ONLY if needed — prefer a plain trait
with `-> impl Future` or a concrete method returning a boxed future; simplest is an `async fn` in a
trait, which is stable on edition 2024). Suggested shape:
```rust
pub trait HttpClient {
    async fn get_json(&self, url: &str, query: &[(&str, &str)]) -> miette::Result<String>;
}

pub struct ReqwestClient { inner: reqwest::Client }

impl ReqwestClient {
    pub fn new() -> miette::Result<Self> {
        let inner = reqwest::Client::builder()
            .user_agent("FerrisFilter/0.1 (github.com/Jonathas-Conceicao/poe2_myfilter; contact: jonathasaoc@gmail.com)")
            .timeout(std::time::Duration::from_secs(30))
            .build()
            .map_err(|e| miette::miette!("Cannot build HTTP client: {e}"))?;
        Ok(Self { inner })
    }
}

impl HttpClient for ReqwestClient {
    async fn get_json(&self, url: &str, query: &[(&str, &str)]) -> miette::Result<String> {
        self.inner.get(url).query(query).send().await
            .map_err(|e| miette::miette!("request to {url} failed: {e}"))?
            .error_for_status()
            .map_err(|e| miette::miette!("{url} returned HTTP error: {e}"))?
            .text().await
            .map_err(|e| miette::miette!("reading body from {url} failed: {e}"))
    }
}
```
(If `async fn` in trait triggers a dyn-compatibility issue downstream, plans 02/03 take a generic
